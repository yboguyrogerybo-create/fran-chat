const express = require('express'); const app = express(); const http = require('http').createServer(app); const io = require('socket.io')(http, { maxHttpBufferSize: 1e7 }); app.use(express.static(__dirname)); app.get('/', (req,res)=>res.sendFile(__dirname+'/index.html')); let users = {}; io.on('connection', socket=>{ socket.on('join', data=>{ socket.myName = data.name || 'Invité'; users[socket.id] = socket.myName; io.emit('online', Object.values(users)); }); socket.on('chat message', msg=>{ io.emit('chat message', msg); }); socket.on('disconnect', ()=>{ delete users[socket.id]; io.emit('online', Object.values(users)); }); }); http.listen(process.env.PORT || 3000);
